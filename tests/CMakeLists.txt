find_package(Qt6 REQUIRED COMPONENTS Test)

function(plainrun_add_test name)
    qt_add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE plainrun_core Qt6::Test ${ARGN})
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;LC_ALL=C.UTF-8;LANG=C.UTF-8")
endfunction()

plainrun_add_test(tst_runformat)
plainrun_add_test(tst_validation)
